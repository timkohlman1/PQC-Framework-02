cmake_minimum_required(VERSION 3.22)
project(pqc_host_wolfmqtt_validation C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

get_filename_component(PQC_REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PQC_WOLFMQTT_SOURCE_DIR "${PQC_REPOSITORY_ROOT}/Middlewares/Third_Party/wolfMQTT" CACHE PATH "wolfMQTT source tree")
set(PQC_WOLFSSL_SOURCE_DIR "${PQC_REPOSITORY_ROOT}/tools/wolfboot/lib/wolfssl" CACHE PATH "wolfSSL source tree")

foreach(required_path
    "${PQC_WOLFMQTT_SOURCE_DIR}/src/mqtt_client.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/examples/mqttclient/mqttclient.c"
    "${PQC_WOLFSSL_SOURCE_DIR}/CMakeLists.txt")
    if(NOT EXISTS "${required_path}")
        message(FATAL_ERROR "Required Phase M9 source is unavailable: ${required_path}")
    endif()
endforeach()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WOLFSSL_INSTALL no CACHE STRING "" FORCE)
set(WOLFSSL_EXAMPLES no CACHE STRING "" FORCE)
set(WOLFSSL_CRYPT_TESTS no CACHE STRING "" FORCE)
set(WOLFSSL_TLS13 yes CACHE STRING "" FORCE)
set(WOLFSSL_SINGLE_THREADED yes CACHE STRING "" FORCE)
add_subdirectory("${PQC_WOLFSSL_SOURCE_DIR}" wolfssl-build EXCLUDE_FROM_ALL)

# The pinned wolfMQTT headers normally include wolfmqtt/options.h generated by
# wolfMQTT's Autotools/CMake build. This project compiles the pinned sources
# directly, so provide an explicit host user-settings boundary instead of
# depending on a generated file in the source tree.
set(PQC_WOLFMQTT_GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${PQC_WOLFMQTT_GENERATED_INCLUDE_DIR}")
file(WRITE "${PQC_WOLFMQTT_GENERATED_INCLUDE_DIR}/user_settings.h"
    "#ifndef PQC_HOST_WOLFMQTT_USER_SETTINGS_H\n"
    "#define PQC_HOST_WOLFMQTT_USER_SETTINGS_H\n"
    "/* Phase M9 host build settings are supplied by target definitions. */\n"
    "#endif\n")

add_executable(pqc_host_wolfmqtt_client
    "${PQC_WOLFMQTT_SOURCE_DIR}/src/mqtt_client.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/src/mqtt_packet.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/src/mqtt_socket.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/examples/mqttexample.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/examples/mqttnet.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/examples/mqttport.c"
    "${PQC_WOLFMQTT_SOURCE_DIR}/examples/mqttclient/mqttclient.c"
)

target_include_directories(pqc_host_wolfmqtt_client PRIVATE
    "${PQC_WOLFMQTT_GENERATED_INCLUDE_DIR}"
    "${PQC_WOLFMQTT_SOURCE_DIR}"
    "${PQC_WOLFSSL_SOURCE_DIR}"
)

target_compile_definitions(pqc_host_wolfmqtt_client PRIVATE
    BUILDING_WOLFMQTT
    WOLFMQTT_USER_SETTINGS
    ENABLE_MQTT_TLS
    HAVE_SOCKET
    HAVE_SIGNAL
)

target_link_libraries(pqc_host_wolfmqtt_client PRIVATE wolfssl::wolfssl)

if(UNIX AND NOT APPLE)
    target_link_libraries(pqc_host_wolfmqtt_client PRIVATE m)
endif()
